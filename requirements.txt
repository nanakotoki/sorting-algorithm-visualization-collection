# ---- 必需 ----------------------------------------------------------------
# GUI 框架（Qt6 官方 Python 绑定）。程序全程只用 QtWidgets / QtGui / QtCore，
# 未使用 QtWebEngine、QtQuick 等重型模块，打包时也已显式排除。
PySide6>=6.6,<7
# 渲染内核：全部几何计算、柱体填充、点阵批绘都是 NumPy 向量化实现
numpy>=1.24

# ---- 可选（功能增强，缺失自动降级）--------------------------------------
# 实时音效输出（PortAudio）。缺失时「音效」开关会自动置灰。
sounddevice>=0.4.6
# 离线 WAV 音轨写出（视频导出的音轨优先走 scipy.io.wavfile）
scipy>=1.10

# ---- 打包（仅 build_exe.py 需要）----------------------------------------
pyinstaller>=6.0
